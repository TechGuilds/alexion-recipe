/* Local mock navigation. Remove public/mock-preview and the marked root layout script to uninstall. */
(() => {
  if (document.getElementById('kajoo-mock-preview')) return;
  const host = document.createElement('aside');
  host.id = 'kajoo-mock-preview';
  host.setAttribute('aria-label', 'Mock preview navigation');
  const shadow = host.attachShadow({ mode: 'open' });
  const style = document.createElement('style');
  style.textContent = `
    :host{position:fixed;top:12px;left:12px;z-index:2147483000;font:14px/1.5 system-ui,sans-serif;color:#172033;text-align:left}
    *{box-sizing:border-box}button{font:inherit;cursor:pointer;background:#172033;color:white;border:0;border-radius:8px;padding:9px 13px;box-shadow:0 2px 12px #0003}
    nav{margin-top:6px;width:min(300px,calc(100vw - 24px));max-height:65vh;overflow:auto;background:white;border:1px solid #ccd3de;border-radius:9px;box-shadow:0 6px 24px #0003;padding:12px}
    nav[hidden]{display:none}p{margin:0 0 8px;color:#536078;font-size:12px}ul{list-style:none;margin:0;padding-left:15px}nav>ul{padding-left:0}
    li{margin:4px 0}summary{cursor:pointer;color:#536078;overflow-wrap:anywhere}a{color:#204a9a;display:block;padding:3px 6px;border-radius:4px;text-decoration:none;overflow-wrap:anywhere}a:hover{background:#edf2fa}a[aria-current=page]{background:#e3edff;font-weight:600}
  `;
  const button = document.createElement('button');
  button.type = 'button'; button.textContent = 'Preview pages';
  button.setAttribute('aria-expanded', 'true'); button.setAttribute('aria-controls', 'pages');
  const nav = document.createElement('nav'); nav.id = 'pages'; nav.setAttribute('aria-label', 'Preview pages');
  nav.textContent = 'Loading preview pages…';
  button.onclick = () => { nav.hidden = !nav.hidden; button.setAttribute('aria-expanded', String(!nav.hidden)); };
  shadow.addEventListener('keydown', (event) => { if (event.key === 'Escape') { nav.hidden = true; button.setAttribute('aria-expanded', 'false'); button.focus(); } });
  shadow.append(style, button, nav); document.body.append(host);
  // Normalize each segment before assigning href; encoded separators and dot paths are invalid too.
  const safeRoute = (value) => {
    if (typeof value !== 'string' || !value.startsWith('/') || value.startsWith('//')) return null;
    if (value === '/') return '/';
    try {
      const segments = value.slice(1).split('/').map(decodeURIComponent);
      if (segments.some((part) => !part || part === '.' || part === '..' || /[\\/?#%:\x00-\x20\x7f]/.test(part))) return null;
      return '/' + segments.map(encodeURIComponent).join('/');
    } catch { return null; }
  };
  fetch('/mock-preview/routes.json', { cache: 'no-store', redirect: 'error', credentials: 'same-origin' }).then((response) => {
    if (!response.ok) throw new Error('Route list unavailable');
    return response.json();
  }).then((data) => {
    if (!data || !Array.isArray(data.routes)) throw new Error('Invalid route list');
    const routes = [...new Set(data.routes.map(safeRoute).filter(Boolean))].sort();
    const root = { children: new Map() };
    for (const route of routes) {
      let node = root;
      for (const segment of route.split('/').filter(Boolean)) {
        if (!node.children.has(segment)) node.children.set(segment, { children: new Map() });
        node = node.children.get(segment);
      }
      node.route = route;
    }
    const current = safeRoute(location.pathname.replace(/\/$/, '') || '/');
    const link = (route, label) => {
      const anchor = document.createElement('a'); anchor.href = route; anchor.textContent = label;
      if (current === route) anchor.setAttribute('aria-current', 'page');
      return anchor;
    };
    const tree = (node) => {
      const list = document.createElement('ul');
      for (const [name, child] of node.children) {
        const item = document.createElement('li'); const label = decodeURIComponent(name).replace(/-/g, ' ');
        if (child.children.size) {
          const details = document.createElement('details'); details.open = true;
          const summary = document.createElement('summary'); summary.textContent = label;
          details.append(summary);
          if (child.route) details.append(link(child.route, 'Open page'));
          details.append(tree(child)); item.append(details);
        } else if (child.route) item.append(link(child.route, label));
        list.append(item);
      }
      return list;
    };
    const info = document.createElement('p'); info.textContent = `${routes.length} preview pages · local preview`;
    nav.replaceChildren(info);
    if (root.route) nav.append(link('/', 'Home'));
    nav.append(tree(root));
    if (!routes.length) info.textContent = 'No preview pages yet. Prepare a new preview stage, then reload.';
  }).catch(() => { nav.textContent = 'Preview page list unavailable. Refresh captures and reload.'; });
})();
